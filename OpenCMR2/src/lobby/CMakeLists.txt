if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.19)
    project(opencmr2_lobby LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
        set(CMAKE_BUILD_TYPE Release)
    endif()
    option(OPENCMR2_BUILD_TESTS "Build the lobby server test" ON)
    set(OPENCMR2_LOBBY_STANDALONE ON)
else()
    set(OPENCMR2_LOBBY_STANDALONE OFF)
endif()

find_package(Threads REQUIRED)

add_library(opencmr2_lobby STATIC lobby_server.cpp)
target_include_directories(opencmr2_lobby PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_compile_features(opencmr2_lobby PUBLIC cxx_std_17)
target_compile_options(opencmr2_lobby PRIVATE -Wall -Wextra)

add_executable(opencmr2-lobby main.cpp)
target_link_libraries(opencmr2-lobby PRIVATE opencmr2_lobby)
target_compile_options(opencmr2-lobby PRIVATE -Wall -Wextra)

if(OPENCMR2_LOBBY_STANDALONE)
    include(GNUInstallDirs)
    install(TARGETS opencmr2-lobby RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()

set(OPENCMR2_LOBBY_TEST_SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/lobby_server.cpp)
if(OPENCMR2_BUILD_TESTS AND NOT ANDROID AND EXISTS ${OPENCMR2_LOBBY_TEST_SOURCE})
    add_executable(lobby_server_test ${OPENCMR2_LOBBY_TEST_SOURCE})
    target_link_libraries(lobby_server_test PRIVATE opencmr2_lobby Threads::Threads)
    target_compile_options(lobby_server_test PRIVATE -Wall -Wextra)
    if(OPENCMR2_LOBBY_STANDALONE)
        enable_testing()
        add_test(NAME lobby_server COMMAND lobby_server_test)
        set_tests_properties(lobby_server PROPERTIES TIMEOUT 60)
    endif()
endif()
